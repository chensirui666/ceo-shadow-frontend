- Preserve node positions while filtering; use semantic source colour CSS tokens and an accessible live text summary.
- Keep all Chinese and English visible copy in `src/content/translations.ts`.

- [ ] **Step 3: Implement the minimum pure state module**

```ts
export type MemoryLayer = 'context' | 'user'
export type MemorySource = 'all' | 'dingtalk' | 'feishu' | 'teams' | 'file' | 'conversation'
export type MemoryNodeSource = Exclude<MemorySource, 'all'>
export type MaterialKind = 'file' | 'migration'

export const visibleMemoryGraph = (graph: MemoryGraphData, query: MemoryQuery): MemoryGraphData => {
  const keyword = query.keyword.trim().toLocaleLowerCase()
  const nodes = graph.nodes.filter((node) => node.layer === query.layer
    && (query.source === 'all' || node.source === query.source)
    && (!keyword || `${node.title} ${node.summary}`.toLocaleLowerCase().includes(keyword)))
  const ids = new Set(nodes.map((node) => node.id))
  return { nodes, edges: graph.edges.filter((edge) => ids.has(edge.from) && ids.has(edge.to)) }
}

export const startMaterialTask = (kind: MaterialKind, fileName: string, size: number): MaterialTask => ({
  kind, fileName, status: 'processing', outcome: size === 0 ? 'failed' : 'completed',
})

export const finishMaterialTask = (task: MaterialTask): MaterialTask => ({ ...task, status: task.outcome })

export const resolveMaterialTask = (graph: MemoryGraphData, task: MaterialTask): MemoryGraphData => (
  task.status === 'failed' ? graph : { nodes: [...graph.nodes, materialNode(task)], edges: [...graph.edges, materialEdge(task)] }
)
```

`MaterialTask` contains a `status: 'processing' | 'completed' | 'failed'` and separate `outcome: 'completed' | 'failed'`; `startMaterialTask` always starts at `processing`, and empty selected files have the failed outcome. Use a small, curated initial graph with all five sources represented in the context layer, cross-source edges, and a separate user layer. The exact fixture node ids must be stable because Task 2 uses them for positions.

- [ ] **Step 1: Build the SVG renderer and scoped scene styling**

```tsx
type MemoryGraphProps = {
  edges: MemoryEdge[]
  nodes: MemoryNode[]
  summary: string
}

const positions: Record<string, { x: number; y: number }> = {
  'context-brief': { x: 432, y: 275 },
  'context-roadmap': { x: 292, y: 172 },
  'context-team': { x: 655, y: 150 },
}

const fallbackPosition = (id: string) => {
  const hash = [...id].reduce((value, character) => (value * 31 + character.charCodeAt(0)) % 997, 0)
  return { x: 150 + hash % 700, y: 105 + (hash * 7) % 350 }
}
const positionFor = (id: string) => positions[id] ?? fallbackPosition(id)
const edgeCoordinates = (edge: MemoryEdge) => {
  const from = positionFor(edge.from)
  const to = positionFor(edge.to)
  return { x1: from.x, x2: to.x, y1: from.y, y2: to.y }
}

export default function MemoryGraph({ edges, nodes, summary }: MemoryGraphProps) {
  return <section aria-label={summary} className="memory-graph">
    <p aria-live="polite" className="sr-only">{summary}</p>
    <svg aria-hidden="true" viewBox="0 0 1000 560">
      {edges.map((edge) => <line key={`${edge.from}-${edge.to}`} {...edgeCoordinates(edge)} className="memory-edge" />)}
      {nodes.map((node) => {
        const position = positionFor(node.id)
        return <g className={`memory-node memory-node-${node.source} memory-node-${node.tier}`} key={node.id} transform={`translate(${position.x} ${position.y})`}>
          <circle r={node.tier === 'core' ? 16 : node.tier === 'connection' ? 10 : 6} />
          {node.tier === 'core' && <text x={22} y={5}>{node.title}</text>}
        </g>
      })}
    </svg>
  </section>
}
```

Add `--memory-source-*` and `--memory-edge` CSS variables once under `.memory-page`. Make the SVG responsive, draw edges before nodes, label only core nodes, and use `@media (prefers-reduced-motion: reduce)` to remove the opacity transition.
